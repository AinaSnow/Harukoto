cmake_minimum_required(VERSION 3.16)
project(MeowpuutWindows LANGUAGES CXX)
if(MSVC)
    # The installed TSF DLL must not require a separately installed VC runtime.
    set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
add_subdirectory(../Core core)
if(WIN32)
    add_library(MeowpuutTip SHARED TextService.cpp MeowpuutTip.def)
    target_link_libraries(MeowpuutTip PRIVATE MeowpuutCore ole32 oleaut32 uuid advapi32 gdi32 user32)
    target_compile_features(MeowpuutTip PRIVATE cxx_std_17)
    if(MSVC)
        target_compile_options(MeowpuutTip PRIVATE /utf-8 /W4)
    endif()
endif()
add_library(MeowpuutWindowsSession STATIC Session.cpp Keyboard.cpp)
target_link_libraries(MeowpuutWindowsSession PUBLIC MeowpuutCore)
target_compile_features(MeowpuutWindowsSession PUBLIC cxx_std_17)
if(MSVC)
    target_compile_options(MeowpuutWindowsSession PRIVATE /utf-8 /W4)
endif()
include(CTest)
if(BUILD_TESTING)
    add_executable(harukoto_conjugation_panel_test tests/ConjugationPanelTests.cpp)
    target_link_libraries(harukoto_conjugation_panel_test PRIVATE MeowpuutCore)
    target_compile_features(harukoto_conjugation_panel_test PRIVATE cxx_std_17)
    if(MSVC)
        target_compile_options(harukoto_conjugation_panel_test PRIVATE /utf-8 /UNDEBUG)
    else()
        target_compile_options(harukoto_conjugation_panel_test PRIVATE -UNDEBUG)
    endif()
    add_executable(harukoto_conjugation_test tests/JapaneseConjugationTests.cpp)
    target_compile_features(harukoto_conjugation_test PRIVATE cxx_std_17)
    target_link_libraries(harukoto_conjugation_test PRIVATE MeowpuutCore)
    if(MSVC)
        target_compile_options(harukoto_conjugation_test PRIVATE /utf-8 /UNDEBUG)
    else()
        target_compile_options(harukoto_conjugation_test PRIVATE -UNDEBUG)
    endif()
    add_test(NAME japanese_conjugations COMMAND harukoto_conjugation_test)
    if(WIN32)
        add_executable(harukoto_japanese_integration_test tests/JapaneseCandidateIntegrationTests.cpp)
        target_link_libraries(harukoto_japanese_integration_test PRIVATE MeowpuutCore)
        target_compile_features(harukoto_japanese_integration_test PRIVATE cxx_std_17)
        if(MSVC)
            target_compile_options(harukoto_japanese_integration_test PRIVATE /utf-8 /UNDEBUG)
        endif()
        add_executable(harukoto_script_preference_test tests/ScriptPreferenceTests.cpp)
        target_link_libraries(harukoto_script_preference_test PRIVATE advapi32)
        target_compile_features(harukoto_script_preference_test PRIVATE cxx_std_17)
        if(MSVC)
            target_compile_options(harukoto_script_preference_test PRIVATE /utf-8 /UNDEBUG)
        endif()
        add_test(NAME script_preference COMMAND harukoto_script_preference_test)
    endif()
    add_executable(meowpuut_session_test tests/session_test.cpp)
    target_link_libraries(meowpuut_session_test PRIVATE MeowpuutWindowsSession)
    add_executable(meowpuut_mixed_candidate_test tests/MixedCompositionTests.cpp)
    target_link_libraries(meowpuut_mixed_candidate_test PRIVATE MeowpuutCore)
    add_test(NAME mixed_candidates COMMAND meowpuut_mixed_candidate_test)
    add_executable(meowpuut_pending_delivery_test tests/PendingDeliveryTests.cpp)
    add_test(NAME pending_delivery COMMAND meowpuut_pending_delivery_test)
    add_executable(meowpuut_punctuation_test tests/PinyinPunctuationTests.cpp)
    add_test(NAME pinyin_punctuation COMMAND meowpuut_punctuation_test)
    add_executable(meowpuut_script_mode_test tests/ScriptModeTests.cpp)
    target_link_libraries(meowpuut_script_mode_test PRIVATE MeowpuutCore)
    add_test(NAME script_mode COMMAND meowpuut_script_mode_test)
    if(MSVC)
        foreach(test_target meowpuut_session_test meowpuut_mixed_candidate_test meowpuut_pending_delivery_test meowpuut_punctuation_test meowpuut_script_mode_test)
            target_compile_options(${test_target} PRIVATE /UNDEBUG)
        endforeach()
        target_compile_options(meowpuut_session_test PRIVATE /utf-8 /W4)
        target_compile_options(meowpuut_mixed_candidate_test PRIVATE /utf-8 /W4)
        target_compile_options(meowpuut_pending_delivery_test PRIVATE /utf-8 /W4)
        target_compile_options(meowpuut_punctuation_test PRIVATE /utf-8 /W4)
        target_compile_options(meowpuut_script_mode_test PRIVATE /utf-8 /W4)
    endif()
    set(MEOWPUUT_LEXICON_DIR "" CACHE PATH "Directory containing native-pinyin.tsv, native-zhuyin.tsv and native-japanese.tsv")
    if(MEOWPUUT_LEXICON_DIR)
        if(EXISTS "${MEOWPUUT_LEXICON_DIR}/japanese-conjugations.bin")
            if(WIN32 AND EXISTS "${MEOWPUUT_LEXICON_DIR}/language-evidence.sqlite")
                add_test(NAME japanese_candidates_production COMMAND harukoto_japanese_integration_test "${MEOWPUUT_LEXICON_DIR}")
            endif()
            add_test(NAME japanese_conjugation_panel COMMAND harukoto_conjugation_panel_test "${MEOWPUUT_LEXICON_DIR}/japanese-conjugations.bin")
            add_test(NAME japanese_conjugations_real COMMAND harukoto_conjugation_test "${MEOWPUUT_LEXICON_DIR}/japanese-conjugations.bin")
        endif()
        add_test(NAME session COMMAND meowpuut_session_test "${MEOWPUUT_LEXICON_DIR}/native-pinyin.tsv" "${MEOWPUUT_LEXICON_DIR}/native-zhuyin.tsv" "${MEOWPUUT_LEXICON_DIR}/native-japanese.tsv")
        add_test(NAME mixed_candidates_real COMMAND meowpuut_mixed_candidate_test "${MEOWPUUT_LEXICON_DIR}/native-pinyin.tsv")
        if(EXISTS "${MEOWPUUT_LEXICON_DIR}/japanese.tsv")
            add_test(NAME japanese_bridge_real COMMAND meowpuut_mixed_candidate_test "${MEOWPUUT_LEXICON_DIR}/native-pinyin.tsv" "${MEOWPUUT_LEXICON_DIR}/japanese.tsv")
        endif()
    endif()
endif()
